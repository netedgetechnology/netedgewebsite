
/* 
This function returns the userid of a user.
Input Parameters: a string representing the username. It is case insensitive.
Returns the userid if the username was found. If not it returns a -1;
*/
CREATE FUNCTION f_get_user_id(p_userName varchar(50)) RETURNS BIGINT
    DETERMINISTIC
BEGIN
    DECLARE l_userID BIGINT;
	
	SET userID = -1;
	
	SELECT user_id INTO l_userID 
	FROM users 
	WHERE TRIM(LOWER(user_name)) = TRIM(LOWER(p_userName));
	
	RETURN (l_userID);
END


/* 
This function verifies if the user login information is valid. It verifies the password for the given user name.
Input Parameters: userName and Password
This procedure requires 2 out parameters to which it will save the user role and user_id. If the username is not found or if the password 
that was provided did not match what is saved for the username, the user_id that is returned is -1.
*/
delimiter //
CREATE PROCEDURE p_verify_login_info(IN p_userName varchar(50),
IN p_password varchar(50), 
OUT p_role varchar(50),
OUT p_user_id BIGINT)
BEGIN
	
	DECLARE l_password VARCHAR(50) DEFAULT "";
	DECLARE l_role VARCHAR(50) DEFAULT "";
	DECLARE l_user_id BIGINT DEFAULT -1;	 
		
	SELECT user_id, password, role 
	INTO l_user_id, l_password, l_role 
	FROM users 
	WHERE TRIM(LOWER(user_name)) = TRIM(LOWER(p_userName));
	
	IF p_password != l_password then
		SET l_user_id = -1;
		SET l_role = "";
	END IF
	
	SET p_role = l_role;
	SET p_user_id = l_user_id; 
END//
delimiter ;



CREATE PROCEDURE p_get_user_info(IN p_userID BIGINT)
BEGIN	
	SELECT * 
	FROM users 
	WHERE user_id = p_userID;
END


/* 
This function registers a user with the given information.
It Checks if the username or email already exists. If they do, then the user information is not saved and a comment is returned stating such.
This procedure requires 2 out parameters to which it will save the userID and the comments. 
If the user information was not saved, then the user_id that is returned is -1 else the new user_id of the user.
*/
CREATE PROCEDURE p_register_user(IN p_firstname varchar(50),
IN p_lastname varchar(50), 
IN p_email varchar(50),
IN p_phone varchar(50),
IN p_userName varchar(50),
IN p_password varchar(50),
OUT p_user_id BIGINT,
OUT p_comment varchar(100))
BEGIN
	
	DECLARE l_username VARCHAR(50) DEFAULT "";
	DECLARE l_email VARCHAR(50) DEFAULT "";
	DECLARE l_comment VARCHAR(100) DEFAULT "";
	DECLARE l_user_id BIGINT DEFAULT -1;	 
		
	SELECT user_name  
	INTO l_username
	FROM users 
	WHERE TRIM(LOWER(user_name)) = TRIM(LOWER(p_userName));
	
	SELECT email   
	INTO l_email
	FROM users 
	WHERE TRIM(LOWER(email)) = TRIM(LOWER(p_email));	
	
	IF l_username != "" then
		SET l_comment = "The Username Already Exists. Please use a different username.";		
	ELSIF l_email != "" then
		SET l_comment = "The Email Already Exists. Please use a different email or log in using the account associated with this email.";
	ELSE		
		INSERT INTO users(first_name, last_name, email, user_name, password, phone, role) 
		VALUES(p_firstname, p_lastname, p_email, p_userName, p_password, p_phone, "user");
		
		SELECT user_id 
		INTO l_user_id 
		FROM users
		WHERE TRIM(LOWER(user_name)) = TRIM(LOWER(p_userName));	

		SET l_comment = "Successfully Registered";
	END IF
	
	SET p_user_id = l_user_id;
	SET p_comment = l_comment;	 
END
